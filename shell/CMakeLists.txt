cmake_minimum_required(VERSION 3.21)
project(WinPebblePageDuplicatorShell LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 MSVC build required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE)
add_compile_options(/W4 /WX /utf-8 /EHsc)
add_library(PageDuplicatorShell SHARED ExplorerCommand.cpp ExplorerCommand.def)
set_target_properties(PageDuplicatorShell PROPERTIES OUTPUT_NAME "WinPebble.PDFPageDuplicator.Shell")
target_link_libraries(PageDuplicatorShell PRIVATE shell32 shlwapi ole32 uuid)

include(CTest)
if(BUILD_TESTING)
  add_executable(ShellCommandTests ../tests/shell_command_test.cpp)
  target_include_directories(ShellCommandTests PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(ShellCommandTests PRIVATE shell32 shlwapi ole32 uuid)
  add_executable(ArgumentRecorder ../tests/shell_argument_recorder.cpp)
  # Test-only fake UI beside the test DLL; never copied into the real package.
  set_target_properties(ArgumentRecorder PROPERTIES OUTPUT_NAME "WinPebble.PDFPageDuplicator")
  add_test(NAME ExplorerCommandSelectionAndLaunch COMMAND ShellCommandTests $<TARGET_FILE:PageDuplicatorShell>)
endif()
